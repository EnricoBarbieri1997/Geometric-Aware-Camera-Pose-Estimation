\section{Appendix B: Algorithms}

% Algorithm: Homotopy Path Tracking (Predictor-Corrector Loop)
\begin{algorithm}[H]
\caption{Homotopy Path Tracking via Predictor-Corrector}
\label{alg:homotopy_path_tracking}
\begin{algorithmic}[1]
\REQUIRE Initial solution $x_0$, initial time $t_0 = 0$, step size $\Delta t$
\STATE $k \leftarrow 0$
\STATE $x_k \leftarrow x_0$, $t_k \leftarrow t_0$
\WHILE{$t_k < 1$}
    \STATE Compute $\dot x_k = -\left(\frac{\partial H}{\partial x}(x_k, t_k)\right)^{-1} \frac{\partial H}{\partial t}(x_k, t_k)$
    \STATE $\tilde x \leftarrow x_k + \Delta t\,\dot x_k$ \; (Predictor: Euler or higher order)
    \STATE $t_{k+1} \leftarrow t_k + \Delta t$
    \STATE $x_{k+1} \leftarrow$ Newton's method to solve $H(x, t_{k+1})=0$ starting from $\tilde x$
    \STATE $k \leftarrow k+1$
\ENDWHILE
\RETURN $x_k$ (approximate solution to $F(x)=0$)
\end{algorithmic}
\end{algorithm}

% Function: Project a point onto a line
%
\begin{algorithm}[H]
\caption{Project Point Onto Line}
\label{alg:project_point_onto_line}
\begin{algorithmic}[1]
\REQUIRE Point to project $p$, line with unitary direction vector $d$ and origin $o$
\STATE
\[ p_{\text{proj}} = o + \langle d, p - o\rangle d \]
\RETURN $p_{\text{proj}}$
\end{algorithmic}
\end{algorithm}

% Algorithm: Get Tangent Points on a Circle
%
\begin{algorithm}[H]
\caption{Calculate tangent points on a circle from a reference point $p$}
\label{alg:tangent_points_circle_point}
\begin{algorithmic}[1]
\REQUIRE Circle center $c$ normal $n$ and radius $r$, point $p$
\STATE $l = p - c$
\STATE $d = |l|$
\STATE $R = \sqrt{d^2 - r^2}$
\STATE $\rho = r/d$
\STATE $\var{ad} = \rho^2 * d$
\STATE $\var{bd} = \rho * R / d * d$
\STATE $\var{adv} = l / d$
\STATE $\var{bdv} = \var{normalize}(n \times \var{adv})$
\STATE $T_1 = c + \var{ad} * \var{adv} + \var{bd} * \var{bdv}$
\STATE $T_2 = c + \var{ad} * \var{adv} - \var{bd} * \var{bdv}$
\RETURN $T_1, T_2$
\end{algorithmic}
\end{algorithm}

\begin{algorithm} [H]
  \caption{Find lines in homogeneous coordinates from a cylinder and a camera}
  \label{alg:get_cylinder_contours}
\begin{algorithmic}[1]
\REQUIRE Camera center $\mathbf{t}$, camera matrix $\PP$, cylinder center $\mathbf{c}$, axis $axis$, radius $r$. 
\STATE $\var{circlecenter} = \var{project\_point\_into\_line}(\mathbf{t}, \var{Line}(\mathbf{c}, \var{axis}))$
\STATE $\var{tangentpoint_1}, \var{tangentpoint_2} = \var{get\_tangentpoints\_circle\_point}(
	\var{Circle}(\var{circlecenter}, r, \var{axis}),
	\mathbf{t}
)$
\STATE $\var{projected\_tangentpoint_1} = \PP \left[\var{tangentpoint_1} \;| \; 1\right]$
\STATE $\var{projected\_tangentpoint_2} = \PP \left[\var{tangentpoint_2} \;| \; 1\right]$
\STATE $\var{projected\_cylinderaxis} = \PP \left[\var{axis} \;| \; 0\right]$
\STATE $\var{contour_1} = \var{projected\_tangentpoint_1} \times ( \var{projected\_tangentpoint_1} + \var{projected\_cylinderaxis})$
\STATE $\var{contour_2} = \var{projected\_tangentpoint_2} \times ( \var{projected\_tangentpoint_2} + \var{projected\_cylinderaxis})$
\RETURN $\var{contour_1} \var{contour_2}$
\end{algorithmic}
\end{algorithm}

% Algorithm: Solution Generation and Homotopy Procedure
\begin{algorithm}[H]
\caption{Solution Generation and Homotopy Procedure}
\label{alg:solutions_generation}
\begin{algorithmic}[1]
\REQUIRE Scene geometry (cylinders), observed silhouette lines $\mathcal{L}_1$, number of cameras $N$
\STATE Initialize intrinsics $\mathbf{K}_0$ (using prior knowledge or default values)
\STATE $\var{best\_solution} \gets \var{null}$
\STATE $\var{best\_solution\_error} \gets \infty$
\STATE $\var{previous\_initialization} \gets \var{null}$
\FOR{$z = 1$ to $Z$}
  \IF{$\var{previous\_initialization}$ is $\var{null}$}
      \STATE Initialize first camera position $\mathbf{t}_0^{(1)}$ randomly
  \ELSE
      \STATE $\mathbf{t}_0^{(1)} = \var{previous\_initialization}$
  \ENDIF
  \FOR{$i = 2$ to $N$}
      \STATE Draw random direction $\mathbf{f}_i$ and distance $d_i$
      \STATE $\mathbf{t}_0^{(i)} = \mathbf{t}_0^{(i-1)} + d_i \mathbf{f}_i$
      \STATE Set rotation $\mathbf{R}_0^{(i)}$ to look at random point $\mathbf{o}$ around world center
  \ENDFOR
  \FOR{each camera $i$}
      \FOR{each cylinder $C_j$}
          \STATE Project $C_j$ to obtain synthetic silhouette $l_{ij}^{h(0)}$ (see Algorithm~\ref{alg:get_cylinder_contours})
      \ENDFOR
  \ENDFOR
  \STATE Construct synthetic system $\mathcal{S}_0$ from all $l_{ij}^{h(0)}$
  \STATE Apply Monodromy to generate additional solutions
  \FOR{each solution $\mathbf{x}_0^{(w)}$ of $\mathcal{S}_0$}
      \STATE Track $\mathbf{x}_0^{(w)}$ through the homotopy $\mathcal{S}_t(\mathbf{x}(t), T_t\mathcal{L}_0)$ from $t=0$ to $t=1$
      \IF{homotopy converges to valid solution $\mathbf{x}_1$}
          \STATE Extract $\mathbf{K}_1$ from $\mathbf{x}_1$
          \FOR{each camera $i$}
              \STATE Extract $\mathbf{R}_1^{(i)}$ from $\mathbf{x}_1$
              \STATE Compute camera position $\mathbf{t}_1^{(i)}$ through a \emph{total degree homotopy} using $\mathcal{L}_1$
          \ENDFOR
          \IF{reprojection error of $\mathbf{x}_1$ is lower than $\var{best\_solution\_error}$}
              \STATE $\var{best\_solution} \gets \mathbf{x}_1$
              \STATE $\var{best\_solution\_error} \gets$ reprojection error of $\mathbf{x}_1$
          \ENDIF
      \ENDIF
  \ENDFOR
  \IF{$\var{best\_solution\_error} < \var{threshold}$}
      \RETURN $\var{best\_solution}$
  \ENDIF
\ENDFOR
\end{algorithmic}
\end{algorithm}
